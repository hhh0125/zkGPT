cmake_minimum_required(VERSION 3.16)
project(zkgpt_paper_r1_foundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
get_filename_component(ZKGPT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(TARGET mcl AND TARGET mclbn256)
set(R1_MCL mcl)
set(R1_MCLBN mclbn256)
else()
find_library(R1_MCL mcl PATHS "${ZKGPT_ROOT}/cmake-build-opt/lib" "${ZKGPT_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
find_library(R1_MCLBN mclbn256 PATHS "${ZKGPT_ROOT}/cmake-build-opt/lib" "${ZKGPT_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
endif()
set(R1_SOURCES lookup_batch.cpp mapped_input.cpp core.cpp pc.cpp sumcheck.cpp memory.cpp range.cpp input.cpp gkr.cpp bundle.cpp wire.cpp integer_relations.cpp matrix.cpp exp.cpp division.cpp positive_sqrt.cpp)
add_library(paper_r1 STATIC ${R1_SOURCES})
target_include_directories(paper_r1 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${ZKGPT_ROOT}/3rd/mcl/include")
target_compile_options(paper_r1 PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(paper_r1 PUBLIC ${R1_MCL} ${R1_MCLBN} pthread)
add_executable(r1_core_tests tests_core.cpp)
target_link_libraries(r1_core_tests PRIVATE paper_r1)
add_test(NAME r1_core COMMAND r1_core_tests)
add_executable(r1_memory_tests tests_memory.cpp)
target_link_libraries(r1_memory_tests PRIVATE paper_r1)
add_test(NAME r1_memory COMMAND r1_memory_tests)
add_executable(r1_range_tests tests_range.cpp)
target_link_libraries(r1_range_tests PRIVATE paper_r1)
add_test(NAME r1_range COMMAND r1_range_tests)
add_executable(r1_input_tests tests_input.cpp)
target_link_libraries(r1_input_tests PRIVATE paper_r1)
add_test(NAME r1_input COMMAND r1_input_tests)
add_executable(r1_gkr_tests tests_gkr.cpp)
target_link_libraries(r1_gkr_tests PRIVATE paper_r1)
add_test(NAME r1_gkr COMMAND r1_gkr_tests)
add_library(paper_r1_verifier STATIC ${R1_SOURCES})
target_include_directories(paper_r1_verifier PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${ZKGPT_ROOT}/3rd/mcl/include")
target_compile_definitions(paper_r1_verifier PRIVATE R1_VERIFY_ONLY=1)
target_compile_options(paper_r1_verifier PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(paper_r1_verifier PUBLIC ${R1_MCL} ${R1_MCLBN} pthread)
add_executable(r1_verify verify_main.cpp)
target_link_libraries(r1_verify PRIVATE paper_r1_verifier)
add_executable(r1_fixtures fixtures_main.cpp)
target_link_libraries(r1_fixtures PRIVATE paper_r1)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME r1_independent COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/independent.py $<TARGET_FILE:r1_fixtures> $<TARGET_FILE:r1_verify>)
set_tests_properties(r1_independent PROPERTIES TIMEOUT 180)
add_executable(r1_integer_tests tests_integer.cpp)
target_link_libraries(r1_integer_tests PRIVATE paper_r1)
add_test(NAME r1_integer COMMAND r1_integer_tests)
add_executable(r1_matrix_tests tests_matrix.cpp)
target_link_libraries(r1_matrix_tests PRIVATE paper_r1)
add_test(NAME r1_matrix COMMAND r1_matrix_tests)
add_executable(r1_exp_tests tests_exp.cpp)
target_link_libraries(r1_exp_tests PRIVATE paper_r1)
add_test(NAME r1_exp COMMAND r1_exp_tests)
add_test(NAME r1_exp_table_certificate COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_exp_table.py --check ${CMAKE_CURRENT_SOURCE_DIR}/exp_table.inc)
add_test(NAME r1_exp_certificate_optimized COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/test_exp_table.py)
add_executable(r1_division_tests tests_division.cpp)
target_link_libraries(r1_division_tests PRIVATE paper_r1)
add_test(NAME r1_division COMMAND r1_division_tests)
add_executable(r1_positive_sqrt_tests tests_positive_sqrt.cpp)
target_link_libraries(r1_positive_sqrt_tests PRIVATE paper_r1)
add_test(NAME r1_positive_sqrt COMMAND r1_positive_sqrt_tests)
set_tests_properties(r1_independent r1_exp_table_certificate r1_exp_certificate_optimized
    PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")

add_executable(r2_mapped_input_tests tests_mapped_input.cpp)
